Task: Mine 3 coal
def mine_3_coal(state, log_fn, step_func):
    from craftax.craftax.constants import Action, BlockType
    # Amounts are added on top of what the player has now
    target = state.inventory.coal + 3
    if state.inventory.pickaxe < 1:
        log_fn("Need a wood pickaxe to mine coal")
        state = explore_until(state, log_fn, step_func, BlockType.TREE, max_steps=40)
        state = mine_block(state, log_fn, step_func, BlockType.TREE, 1, max_steps=30)
        state = craft(state, log_fn, step_func, Action.MAKE_WOOD_PICKAXE)
    for _ in range(3):
        if state.inventory.coal >= target:
            break
        state = explore_until(state, log_fn, step_func, BlockType.COAL, max_steps=60)
        state = mine_block(state, log_fn, step_func, BlockType.COAL, 1, max_steps=30)
    return state
